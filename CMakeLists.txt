cmake_minimum_required(VERSION 3.20)
project(MOEXWatch VERSION 0.6.3 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Pure calculations: no Win32, networking, filesystem or provider adapters.
add_library(moex_domain src/domain_common.cpp src/indicators.cpp src/portfolio_calculations.cpp
  src/signal_rules.cpp src/backtest.cpp src/dividend_calculations.cpp)
target_include_directories(moex_domain PUBLIC include)
enable_testing()
add_executable(domain_tests tests/domain_tests.cpp)
target_link_libraries(domain_tests PRIVATE moex_domain)
if(MSVC)
  target_compile_options(domain_tests PRIVATE /W4 /permissive- /utf-8)
endif()
add_test(NAME domain_tests COMMAND domain_tests)
option(MOEX_DOMAIN_ONLY "Build only portable calculation library and its tests" OFF)
if(MOEX_DOMAIN_ONLY)
  if(MSVC)
    target_compile_options(moex_domain PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(moex_domain PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  endif()
  return()
endif()
add_library(market_core src/market_core.cpp src/dividends.cpp src/storage.cpp src/portfolio_csv.cpp src/observation.cpp)
target_include_directories(market_core PUBLIC include)
target_link_libraries(market_core PUBLIC moex_domain)
if(WIN32)
  target_compile_definitions(market_core PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
endif()

if(MSVC)
  target_compile_options(moex_domain PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(market_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(moex_domain PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  target_compile_options(market_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()

if(WIN32)
  add_library(moex_ui src/ui_shell.cpp src/ui_common.cpp src/ui_backtest.cpp src/ui_signals.cpp src/ui_portfolio.cpp src/ui_trades.cpp src/ui_charts.cpp src/ui_chart_render.cpp src/ui_dividends.cpp src/market_data.cpp src/portfolio_import_ui.cpp src/table_sort_win.cpp src/observation_ui.cpp)
  target_link_libraries(moex_ui PUBLIC market_core winhttp comctl32 gdi32 user32 shell32 comdlg32)
  target_compile_definitions(moex_ui PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MSVC)
    target_compile_options(moex_ui PRIVATE /W4 /permissive- /utf-8)
  endif()
  add_executable(moexwatch WIN32 src/main.cpp)
  target_link_libraries(moexwatch PRIVATE moex_ui)
  if(MSVC)
    target_compile_options(moexwatch PRIVATE /W4 /permissive- /utf-8)
  endif()
  add_executable(ui_pages_tests tests/ui_pages_tests.cpp)
  target_link_libraries(ui_pages_tests PRIVATE moex_ui)
  if(MSVC)
    target_compile_options(ui_pages_tests PRIVATE /W4 /permissive- /utf-8)
  endif()

else()
  add_executable(moexwatch src/main.cpp)
  target_link_libraries(moexwatch PRIVATE market_core)
endif()

enable_testing()
add_executable(market_core_tests tests/market_core_tests.cpp)
target_link_libraries(market_core_tests PRIVATE market_core)
add_test(NAME market_core_tests COMMAND market_core_tests)
add_executable(dividend_tests tests/dividend_tests.cpp)
target_link_libraries(dividend_tests PRIVATE market_core)
target_compile_definitions(dividend_tests PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
if(MSVC)
  target_compile_options(dividend_tests PRIVATE /utf-8)
endif()
add_test(NAME dividend_tests COMMAND dividend_tests)

add_executable(storage_tests tests/storage_tests.cpp)
target_link_libraries(storage_tests PRIVATE market_core)
if(MSVC)
  target_compile_options(storage_tests PRIVATE /utf-8)
endif()
add_test(NAME storage_tests COMMAND storage_tests)

add_executable(portfolio_csv_tests tests/portfolio_csv_tests.cpp)
target_link_libraries(portfolio_csv_tests PRIVATE market_core)
if(MSVC)
  target_compile_options(portfolio_csv_tests PRIVATE /utf-8)
endif()
add_test(NAME portfolio_csv_tests COMMAND portfolio_csv_tests)

add_executable(observation_tests tests/observation_tests.cpp)
target_link_libraries(observation_tests PRIVATE market_core)
if(MSVC)
  target_compile_options(observation_tests PRIVATE /utf-8)
endif()
add_test(NAME observation_tests COMMAND observation_tests)

if(WIN32)
  add_executable(gui_sort_tests tests/gui_sort_tests.cpp src/table_sort_win.cpp src/observation_ui.cpp)
  target_link_libraries(gui_sort_tests PRIVATE market_core comctl32 user32 gdi32)
  target_compile_definitions(gui_sort_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MSVC)
    target_compile_options(gui_sort_tests PRIVATE /W4 /permissive- /utf-8)
  endif()
  add_test(NAME gui_sort_tests COMMAND gui_sort_tests)
endif()

if(WIN32)
  add_test(NAME ui_pages_tests COMMAND ui_pages_tests)
endif()

cmake_minimum_required(VERSION 3.20)
project(MOEXWatch VERSION 0.7.0 LANGUAGES CXX)

option(MOEX_BUILD_QT "Build the Qt Quick investment workspace alongside the classic UI" OFF)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Pure calculations: no Win32, networking, filesystem or provider adapters.
add_library(moex_domain src/domain_common.cpp src/indicators.cpp src/portfolio_calculations.cpp
  src/signal_rules.cpp src/backtest.cpp src/dividend_calculations.cpp src/position_cycles.cpp)
target_include_directories(moex_domain PUBLIC include)
add_library(moex_state src/app_state.cpp)
target_include_directories(moex_state PUBLIC include)
enable_testing()
add_executable(domain_tests tests/domain_tests.cpp)
target_link_libraries(domain_tests PRIVATE moex_domain)
if(MSVC)
  target_compile_options(domain_tests PRIVATE /W4 /permissive- /utf-8)
endif()
add_test(NAME domain_tests COMMAND domain_tests)
add_executable(app_state_tests tests/app_state_tests.cpp)
target_link_libraries(app_state_tests PRIVATE moex_state)
add_test(NAME app_state_tests COMMAND app_state_tests)
if(MSVC)
  target_compile_options(moex_state PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(app_state_tests PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(moex_state PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  target_compile_options(app_state_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_executable(backtest_regression_tests tests/backtest_regression_tests.cpp)
target_link_libraries(backtest_regression_tests PRIVATE moex_domain)
add_test(NAME backtest_regression_tests COMMAND backtest_regression_tests)

option(MOEX_DOMAIN_ONLY "Build portable calculations and application state without UI or adapters" OFF)
if(MOEX_DOMAIN_ONLY)

  endif()
  return()
endif()
find_package(Threads REQUIRED)

target_include_directories(moex_io PUBLIC include)

  target_compile_options(moex_io PRIVATE /W4 /permissive- /utf-8)

  target_compile_options(moex_io PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_library(market_core src/profile_lock.cpp src/app_storage.cpp src/user_storage.cpp src/market_validation.cpp src/market_core.cpp src/dividends.cpp src/storage.cpp src/portfolio_csv.cpp src/observation.cpp src/portfolio_identity.cpp src/portfolio_journal_storage.cpp)

  target_compile_definitions(market_core PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
endif()

if(MSVC)
  target_compile_options(moex_domain PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(market_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(moex_domain PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  target_compile_options(market_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()

if(WIN32)
  add_library(moex_ui src/ui_tasks.cpp src/winhttp_transport.cpp src/ui_shell.cpp src/ui_common.cpp src/ui_backtest.cpp src/ui_signals.cpp src/ui_portfolio.cpp src/ui_trades.cpp src/ui_charts.cpp src/ui_chart_render.cpp src/ui_dividends.cpp src/market_data.cpp src/portfolio_import_ui.cpp src/table_sort_win.cpp src/observation_ui.cpp)
  target_link_libraries(moex_ui PUBLIC market_core winhttp comctl32 gdi32 user32 shell32 comdlg32)
  target_compile_definitions(moex_ui PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MSVC)
    target_compile_options(moex_ui PRIVATE /W4 /permissive- /utf-8)
  endif()
  add_executable(moexwatch WIN32 src/main.cpp)
  target_link_libraries(moexwatch PRIVATE moex_ui)
  if(MSVC)
    target_compile_options(moexwatch PRIVATE /W4 /permissive- /utf-8)
  endif()
  add_executable(ui_pages_tests tests/ui_pages_tests.cpp)
  target_link_libraries(ui_pages_tests PRIVATE moex_ui)
  if(MSVC)
    target_compile_options(ui_pages_tests PRIVATE /W4 /permissive- /utf-8)
  endif()

else()
  add_executable(moexwatch src/main.cpp)
  target_link_libraries(moexwatch PRIVATE market_core)
endif()
